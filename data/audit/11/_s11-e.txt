599: // ── D6 source scan ─────────────────────────────────────────────────────────
600: // Every spring config in beats/*.jsx must be ζ ∈ [0.46, 1.0] EXCEPT the
601: // documented A3.3 / MANUAL D1 E.push exception (ImageBeat's overdamped
602: // settle). Resolves Progress's `config: SPRING_CONFIG` reference to its
603: // literal in the same file. ζ = damping / (2·√(stiffness·mass)), mass = 1
604: // (Remotion default; Easing.spring defaults stiffness 100 when omitted).
605: function zeta(damping, stiffness) {
606:   return damping / (2 * Math.sqrt(stiffness));
607: }

609: function gateD6() {
610:   const dir = resolve(BEATS_DIR);
611:   const files = readdirSync(dir).filter((f) => f.endsWith(".jsx"));
612:   const sites = [];
613:   const configs = {}; // file → named spring config literals
614:   for (const f of files) {
615:     const src = readFileSync(resolve(dir, f), "utf8");
616:     const cfgRe = /const\s+(\w+)\s*=\s*\{([^}]*damping[^}]*)\};?/g;
617:     let cm;
618:     while ((cm = cfgRe.exec(src))) configs[`${f}:${cm[1]}`] = cm[2];
619:     const re = /(?:Easing\.spring|spring)\(\s*\{([^}]*)\}/g;
620:     let m;
621:     while ((m = re.exec(src))) sites.push({ file: f, opts: m[1] });
622:   }
623:   const resolved = sites.map((s) => {
624:     const cfgRef = /config:\s*(\w+)/.exec(s.opts);
625:     const opts = cfgRef ? configs[`${s.file}:${cfgRef[1]}`] || s.opts : s.opts;
626:     const damping = Number(/damping:\s*([\d.]+)/.exec(opts)?.[1] ?? NaN);
627:     const stiffness = Number(/stiffness:\s*([\d.]+)/.exec(opts)?.[1] ?? 100);
628:     return {
629:       ...s,
630:       damping,
631:       stiffness,
632:       zeta: Number.isFinite(damping) ? zeta(damping, stiffness) : NaN,
633:     };
634:   });
635:   const progress = resolved.find((s) => s.file === "Progress.jsx");
636:   const image = resolved.find((s) => s.file === "ImageBeat.jsx");
637:   const others = resolved.filter((s) => s.file !== "Progress.jsx" && s.file !== "ImageBeat.jsx");
638:   const okCount = resolved.length === 2 && others.length === 0;
639:   const okProgress =
640:     progress && Math.abs(progress.zeta - zeta(13.9, 180)) < 1e-6 && progress.zeta >= 0.46 && progress.zeta <= 1.0;
641:   const okImage = image && Math.abs(image.damping - 200) < 1e-9 && image.zeta > 1.0; // the sanctioned exception
642:   return {
643:     pass: okCount && !!okProgress && !!okImage,
644:     note: `sites=${resolved.length} ${resolved
645:       .map((s) => `${s.file} ζ=${s.zeta.toFixed(3)} (d=${s.damping}, k=${s.stiffness})`)
646:       .join(" | ")} — Progress ζ=0.518 in-range, ImageBeat push ζ=10 = A3.3 exception`,
647:   };

653:   const s = w.stage;
654:   const h = w.compiled.headline;
655:   const lefts = [s.x, h ? h.x : s.x];
656:   const tops = [s.y, h ? h.y : s.y];

658:   const bottoms = [s.y + s.h, h ? h.y + h.h : s.y + s.h];
659:   return {
660:     x0: Math.max(0, Math.min(...lefts) - 8),
661:     y0: Math.max(0, Math.min(...tops) - 8),
662:     x1: Math.min(w.width, Math.max(...rights) + 8),
663:     y1: Math.min(w.height, Math.max(...bottoms) + 8),
664:   };
