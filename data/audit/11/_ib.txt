1: import React from "react";
2: import { Easing, interpolate, staticFile, useCurrentFrame } from "remotion";
3: import { D, MG_TYPE } from "../compositions/beats.js";
4: import Layer from "../layers/Layer.jsx";
5: 
6: // ── Motion curves ─ Layer D2/D3 shared values (see Statement.jsx).
7: const E_OUT = Easing.bezier(0.16, 1, 0.3, 1);
8: 
9: function ease(frame, inputRange, outputRange, easing = E_OUT) {
10:   return interpolate(frame, inputRange, outputRange, {
11:     extrapolateLeft: "clamp",
12:     extrapolateRight: "clamp",
13:     easing,
14:   });
15: }
16: 
17: // D2.2 RISE — translateY +24 → 0 over D.base, opacity 0 → 1 over D.short.
18: function riseStyle(frame, start, offsetPx = 24) {
19:   return {
20:     opacity: ease(frame - start, [0, D.short], [0, 1], E_OUT),
21:     translate: `0px ${offsetPx * ease(frame - start, [0, D.base], [1, 0], E_OUT)}px`,
22:   };
23: }
24: 
25: // ── F7 treatment (E2.5/E2.6) — legacy ImageBeatScene (motion-graphics.jsx
26: //    945-992), re-expressed against the stage prop. The stage slot IS the
27: //    image slot (claim 9-07: "48,392,840,548 — the stage slot IS the image
28: //    slot"); every offset derives from the stage rect, no bare layout ints.
29: const IMAGE_RADIUS = 24; // F7 "radius 24"
30: const IMAGE_SATURATE = 0.35; // F7 "saturate(0.35)"
31: const IMAGE_TINT = 0.12; // F7 "accent overlay 0.12 opacity"
32: const CREDIT_INSET_X = 8; // F7 credit at stage.x + 8 (legacy 56 = 48 + 8)
33: const CREDIT_INSET_Y = 34; // F7 credit at stage.y + stage.h − 34 (legacy 906)
34: const PUSH_ENTRY = 1.05; // F7/legacy — entry scale, pushed to 1.00
35: 
36: /**
37:  * IMAGE_BEAT — claim 9-07: the photo fills the stage rect EXACTLY (radius 24,
38:  * overflow hidden), desaturated (saturate 0.35) with a 12 % accent tint
39:  * overlay; fades in over D.base (9 f) from tA−4 at scale 1.05, then pushes
40:  * 1.05 → 1.00 over D.push via Easing.spring({damping: 200}) (legacy: scale
41:  * 1.05 − push·0.05); a 32 px textDim credit line sits at the stage bottom-
42:  * left corner with riseStyle at start+D.short; the subject headline rides the
43:  * compiled rect via Layer (RISE@"anchor+6"); NO accent element (F7 "none" —
44:  * the tint overlay is a style, not an accent element).
45:  * HOLD (A4.1): nothing moves after the push settles tA+56 (start + D.push —
46:  * imageScale reaches 1.00; the fade settles tA+5, the credit rise tA+11,
47:  * the subject headline tA+15). C17 pair (tA+55, tA+56). D6: Easing.spring
48:  * ({damping: 200}) → ζ = 10 ∉ [0.46, 1.0] — the sanctioned A3.3 / MANUAL D1
49:  * E.push exception (overdamped settle, zero bounce).
50:  */
51: export function ImageBeat({
52:   rects = {},
53:   stage,
54:   colors,
55:   anchorFrame = 0,
56:   frame: frameProp,
57:   src = "",
58:   credit = "",
59:   fontFamily = "Inter",
60:   ...rest
61: }) {
62:   const frame = frameProp ?? useCurrentFrame();
63:   const headline = rects.headline;
64:   const start = Math.max((Number(anchorFrame) || 0) - D.micro, 0); // tA−4
65:   const push = Easing.spring({ damping: 200 })(ease(frame - start, [0, D.push], [0, 1], E_OUT));
66:   const imageOpacity = ease(frame - start, [0, D.base], [0, 1], E_OUT);
67:   const imageScale = PUSH_ENTRY - push * 0.05;
68: 
69:   return (
70:     <div {...rest} style={{ position: "absolute", left: 0, top: 0, width: 0, height: 0 }}>
71:       <div
72:         data-role="image"
73:         style={{
74:           position: "absolute",
75:           left: stage.x,
76:           top: stage.y,
77:           width: stage.w,
78:           height: stage.h,
79:           borderRadius: IMAGE_RADIUS,
80:           overflow: "hidden",
81:           opacity: imageOpacity,
82:           scale: `${imageScale}`,
83:           transformOrigin: "center",
84:         }}
85:       >
86:         {src ? (
87:           // Plain <img> — legacy ImageBeatScene (motion-graphics.jsx:945-992)
88:           // uses plain <img> too; Remotion's <Img> throws on load failure,
89:           // which would hard-fail stills when a fixture asset is absent.
90:           <img
91:             src={staticFile(src)}
92:             alt=""
93:             style={{
94:               width: "100%",
95:               height: "100%",
96:               objectFit: "cover",
97:               display: "block",
98:               filter: `saturate(${IMAGE_SATURATE})`,
99:             }}
100:           />
101:         ) : null}
102:         <div
103:           style={{
104:             position: "absolute",
105:             inset: 0,
106:             backgroundColor: colors.accent,
107:             opacity: IMAGE_TINT,
108:           }}
109:         />
110:       </div>
111: 
112:       {credit ? (
113:         <div
114:           data-role="credit"
115:           style={{
116:             position: "absolute",
117:             left: stage.x + CREDIT_INSET_X,
118:             top: stage.y + stage.h - CREDIT_INSET_Y,
119:             fontSize: MG_TYPE.label, // 32 — w400, textDim
120:             fontWeight: 400,
121:             color: colors.textDim,
122:             fontFamily,
123:             whiteSpace: "nowrap",
124:             ...riseStyle(frame, start + D.short),
125:           }}
126:         >
127:           {credit}
128:         </div>
129:       ) : null}
130: 
131:       {headline ? (
132:         <Layer
133:           rect={headline}
134:           enter={headline.enter}
135:           exit={headline.exit}
136:           frame={frame}
137:           data-rect-id={headline.id}
138:           data-role={headline.role}
139:         >
140:           <span
141:             style={{
142:               display: "block",
143:               width: "100%",
144:               height: "100%",
145:               fontSize: headline.fontSize,
146:               lineHeight: `${headline.h}px`,
147:               fontWeight: 800, // MG_TYPE headline — w800
148:               color: colors.textPrimary,
149:               whiteSpace: "nowrap",
150:               textAlign: "center",
151:             }}
152:           >
153:             {headline.text}
154:           </span>
155:         </Layer>
156:       ) : null}
157:     </div>
158:   );
159: }
160: 
161: export default ImageBeat;
