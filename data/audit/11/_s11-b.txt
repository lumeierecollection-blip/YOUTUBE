261: function fixtureMetrics(text, fontSize, slotW) {
262:   const chars = String(text).length;
263:   const charW = 0.6 * fontSize;
264:   const perLine = Math.max(Math.floor(slotW / charW), 1);
265:   const lines = Math.min(Math.ceil(chars / perLine), 2);
266:   return { width: Math.min(chars * charW, slotW), lines };
267: }

269: function buildInputs(fmt) {
270:   const slots = fmt === "shorts" ? SLOTS_SHORTS : SLOTS_LONGFORM;
271:   const safe = fmt === "shorts" ? SAFE_SHORTS : SAFE_LONGFORM;
272:   const fromOpts = {
273:     anchorFrames: Object.fromEntries(FIX.map((b) => [b.id, ANCHOR])),
274:     durations: Object.fromEntries(FIX.map((b) => [b.id, DUR])),
275:     defaultDuration: DUR,
276:     forceAllPersistent: true,
277:     runtime: "runtime 1",
278:     slotTable: slots,
279:     fonts: { family: "Inter" },
280:     measured: {},
281:     fromVersion: "fromBeats-11",
282:   };
283:   const specs = fromBeats(FIX, fromOpts); // ids s0b0..s0b7, FIX order preserved
284:   validateShotSpecs(specs);
285:   const fontFamily = "Inter";
286:   const fonts = {};
287:   const measured = {};
288:   const anchorFrame = {};
289:   specs.forEach((spec, i) => {
290:     const beat = FIX[i];
291:     anchorFrame[spec.id] = Math.max(beat.anchorFrame - beat.startFrame, 0); // 10
292:     spec.layers.forEach((layer, j) => {
293:       const key = `${spec.id}:${j}`;
294:       if (["kicker", "headline", "caption", "support"].includes(layer.role)) {
295:         const fontSize = MG_TYPE[layer.role] || 44;
296:         const text =
297:           layer.role === "kicker"
298:             ? `${String(layer.content.index).padStart(2, "0")} ${layer.content.label}`
299:             : (layer.content.text ?? "");
300:         const slotW = layer.role === "caption" ? slots.caption.w : slots.headline.w;
301:         fonts[key] = { fontSize, fontFamily };
302:         measured[measuredKey(text, fontSize, fontFamily)] = fixtureMetrics(text, fontSize, slotW);
303:       }
304:     });
305:   });
306:   const frames = compile(specs, { slots, safe, fonts, measured, anchorFrame });
307:   return { specs, frames, slots };
308: }

310: function wire(beat, fmt) {
311:   const { specs, frames, slots } = buildInputs(fmt);
312:   const idx = FIX.findIndex((b) => b.id === beat.id);
313:   const spec = specs[idx];
314:   const rects = frames[idx].rects;
315:   const layers = spec.layers || [];
316:   const motionOf = (role) => {
317:     const l = layers.find((x) => x.role === role);
318:     return l ? { enter: l.enter, exit: l.exit } : {};
319:   };
320:   const textOf = (role) => {
321:     const l = layers.find((x) => x.role === role);
322:     if (!l || !l.content) return undefined;
323:     return l.role === "kicker"
324:       ? `${String(l.content.index ?? 0).padStart(2, "0")} ${l.content.label ?? ""}`
325:       : (l.content.text ?? undefined);
326:   };
327:   const headlineLayer = layers.find((x) => x.role === "headline");
328:   const isShort = fmt === "shorts";
329:   const slug = RUN_SLUG[beat.id] || beat.id;
330:   const wired = rects
331:     .filter((r) => !r.structural || r.role === "accent")
332:     .map((r) => {
333:       const t = textOf(r.role);
334:       return {
335:         ...r,
336:         id: `${slug}-${fmt}-${r.role}`,
337:         ...motionOf(r.role),
338:         ...(t !== undefined ? { text: t } : {}),
339:       };
340:     });
341:   return {
342:     id: `${slug}-${fmt}`,
343:     archetype: beat.archetype,
344:     width: isShort ? 1080 : 1920,
345:     height: isShort ? 1920 : 1080,
346:     durationInFrames: DUR,
347:     fps: FPS,
348:     stage: slots.stage,
349:     colors: COLORS,
350:     anchorFrame: tA,
351:     props: archetypeProps(beat),
352:     rects: Object.fromEntries(wired.map((r) => [r.role, r])), // { headline, accent, chart }
353:     compiled: {
354:       headline: wired.find((r) => r.role === "headline") ?? null,
355:       accent: wired.find((r) => r.role === "accent") ?? null,
356:       headlineText: headlineLayer ? headlineLayer.content.text ?? "" : "",
357:       layers,
358:     },
359:   };
360: }
