241: export function assertEqualYearIntervals(series, id) {
242:   if (series.length < 3) return;
243:   const isYear = (label) => {
244:     const s = String(label ?? "").trim();
245:     if (!/^\d{4,}$/.test(s)) return false;
246:     const n = Number(s);
247:     return Number.isInteger(n) && n >= 1000;
248:   };

250:   const years = series.map((p) => Number(p.label));
251:   const deltas = [];

253:   if (deltas.some((d) => d - deltas[0] > 1e-6 || d - deltas[0] < -1e-6)) {
254:     throw new Error(
255:       `${id} PROGRESS: time labels ${years.join(", ")} have unequal intervals ` +
256:         `(${deltas.join(", ")}) â€” a time comparison uses equal intervals or states them (ENC-23, H6)`
257:     );
258:   }
