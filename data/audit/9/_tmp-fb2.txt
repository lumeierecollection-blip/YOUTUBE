109: /** The headline fallback: the beat's own anchor phrase, â‰¤maxChars and â‰¤4
110:  *  tokens (CLAIM-ENC-9-01 â€” CONTRAST's anchor is the first token after the
111:  *  pivot marker, i.e. the consequence start; beats.js pickAnchorTokenIndex). */
112: export function headlineFor(beat, maxChars = HEADLINE_MAX_CHARS, maxTokens = 4) {
113:   const tokens = beat.wordTokens || [];
114:   const anchor = tokens[beat.anchorTokenIndex];
115:   if (!anchor) return String(beat.text || "").split(/\s+/)[0] || "";
116:   let phrase = anchor.text;
117:   let words = 1;
118:   for (let i = beat.anchorTokenIndex + 1; i < tokens.length; i++) {
119:     if (words >= maxTokens) break;
120:     const next = `${phrase} ${tokens[i].text}`;
121:     if (next.length > maxChars) break;
122:     phrase = next;
123:     words += 1;
124:   }
125:   return phrase;
126: }
